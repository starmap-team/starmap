```python
"""Phase 18-01: 跨端一致性抽样测试 (改用 pytest-asyncio).

Fix L4 (review): 仅在 run 已进入 terminal 状态 (completed/failed/cancelled) 时做对比.
避免 mid-run 状态不一致导致误报.
"""
import time
import httpx
import pytest
import asyncio
from sqlalchemy import select, desc
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.session import get_async_engine
from app.models.pipeline_models import PipelineRun

API = "http://localhost:8000/api/v1"


async def _login() -> str:
    async with httpx.AsyncClient() as c:
        r = await c.post(f"{API}/auth/login", json={"username": "admin", "password": "starmap2024"}, timeout=10)
        return r.json()["access_token"]


async def _wait_terminal(run_id: str, max_wait: int = 90) -> dict:
    async with httpx.AsyncClient() as c:
        tok = await _login()
        deadline = time.monotonic() + max_wait
        while time.monotonic() < deadline:
            r = await c.get(f"{API}/pipeline/runs/{run_id}", headers={"Authorization": f"Bearer {tok}"}, timeout=10)
            if r.status_code == 200:
                run = r.json()
                stages = run.get("stages", [])
                if stages and all(s["status"] in ("completed", "failed", "skipped", "cancelled") for s in stages):
                    return run
            await asyncio.sleep(2)
        raise TimeoutError(f"Run {run_id} not terminal after {max_wait}s")


async def _get_latest_run_id() -> str | None:
    engine = get_async_engine()
    async with AsyncSession(bind=engine) as s:
        r = await s.execute(select(PipelineRun).order_by(desc(PipelineRun.started_at)).limit(1))
        run = r.scalar_one_or_none()
        return str(run.id) if run else None


@pytest.mark.parametrize("trial", range(20))
async def test_api_matches_db_per_stage(trial):
    """抽样 N=20: API /pipeline/runs/{run_id} 返回的每个 stage 字段
    必须与 DB PipelineRun.stages JSON 完全一致.
    """
    async with httpx.AsyncClient(timeout=30) as c:
        # 1. 触发新 run
        tok = await _login()
        await c.post(
            f"{API}/pipeline/trigger",
            json={"run_type": "incremental", "selected_stages": ["crawl", "dedup", "clean", "import", "graph_sync"]},
            headers={"Authorization": f"Bearer {tok}"},
            timeout=10,
        )

        # 2. 等待最近 run terminal
        run_id = await _get_latest_run_id()
        if not run_id:
            pytest.skip("No runs found")
        run = await _wait_terminal(run_id, max_wait=90)
        api_stages = run.get("stages", [])

        if not api_stages:
            pytest.skip("API returned no stages")

        # 3. DB 直接查询
        engine = get_async_engine()
        async with AsyncSession(bind=engine) as s:
            r = await s.execute(select(PipelineRun).where(PipelineRun.id == run_id))
            db_run = r.scalar_one()
            if db_run.status == "running":
                pytest.skip(f"Run {run_id} still running")
            db_stages = db_run.stages or []

        # 4. 字段对比
        for api_s in api_stages:
            db_s = next((d for d in db_stages if isinstance(d, dict) and d.get("name") == api_s["name"]), None)
            if db_s is None:
                continue
            assert api_s["name"] == db_s["name"]
            assert api_s["status"] == db_s["status"]
            assert api_s["records_processed"] == db_s.get("records_processed", 0)
            assert abs(api_s.get("progress") or 0 - (db_s.get("progress") or 0)) < 0.01
            assert api_s.get("errors", []) == db_s.get("errors", [])
            assert api_s.get("duration_ms") == db_s.get("duration_ms")
```

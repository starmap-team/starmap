## 目标

`currentRunId` 当前只看 `pipelineStatus?.current_run?.id` (只有 running 才有),改用 `last_run.id` (任何已完成 run 都有)。

```ts
// Phase 17-02: 改用 last_run.id 替代 current_run.id
// 原因: current_run 只在 status=running 时存在,失败的 run 也有可重试 stage
const currentRunId = computed(() => {
  const status = pipeline.pipelineStatus
  return status?.last_run?.id  // 任何 terminal 状态的 run
    ?? status?.current_run?.id  // fallback for running
})
```

## Task 2: 增强 handleRetryStage 错误处理 [frontend]

```ts
async function handleRetryStage(stageName: string) {
  const runId = currentRunId.value
  if (!runId) {
    ElMessage.warning('没有可重试的运行 — 请先触发一次流水线')
    return
  }

  // Phase 17-02: 检查 stage 真的 failed,否则提示用户
  const stage = (pipeline.stages || []).find((s: any) => s.name === stageName)
  if (stage && stage.status !== 'failed' && stage.status !== 'cancelled') {
    ElMessage.info(`阶段「${STAGE_LABELS[stageName] || stageName}」状态为 ${stage.status},无需重试`)
    return
  }

  retryingStages.value.add(stageName)
  try {
    await pipeline.retryStage(runId, stageName)
    await loadAll()
    ElMessage.success(`阶段「${STAGE_LABELS[stageName] || stageName}」已重新调度`)
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : '重试失败'
    ElMessage.error(`重试失败: ${msg}`)
  } finally {
    retryingStages.value.delete(stageName)
  }
}
```

```ts
import { test, expect } from '@playwright/test'

test('重试按钮支持 failed 状态的 run', async ({ page }) => {
  // Mock 失败的 run
  await page.route('**/pipeline/stages', async (route) => {
    await route.fulfill({
      status: 200,
      body: JSON.stringify({
        stages: [
          { name: 'crawl', status: 'completed', records_processed: 100 },
          { name: 'graph_sync', status: 'failed', records_processed: 0, errors: ['extraction_payload missing position_name'] }
        ],
        run_id: 'test-run-001',
        run_status: 'failed'
      })
    })
  })

  await page.goto('/pipeline')
  await page.waitForTimeout(2000)

  // 点击重试按钮 (graph_sync stage)
  const retryBtn = page.locator('button:has-text("重试")').first()
  if (await retryBtn.count() > 0) {
    await retryBtn.click()
    // 不应显示 "没有可重试的运行" 警告
    await page.waitForTimeout(1000)
    const warning = await page.locator('text=没有可重试的运行').count()
    expect(warning).toBe(0)
  }
})
```

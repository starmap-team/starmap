import pytest

@pytest.mark.asyncio
@pytest.mark.parametrize("trial", range(20))
async def test_api_matches_db_per_stage(trial):
    """抽样 N=20: API /pipeline/runs/{run_id} 与 DB 一致."""
    from app.db.session import get_async_engine
    from app.models.pipeline_models import PipelineRun
    from sqlalchemy import select, desc
    from sqlalchemy.ext.asyncio import AsyncSession
    import httpx

    tok_resp = await httpx.AsyncClient().post(
        f"{API}/auth/login", json={"username": "admin", "password": "starmap2024"}, timeout=10
    )
    tok = tok_resp.json()["access_token"]

    # 触发新 run
    await httpx.AsyncClient().post(
        f"{API}/pipeline/trigger",
        json={"run_type": "incremental", "selected_stages": ["crawl", "dedup", "clean", "import", "graph_sync"]},
        headers={"Authorization": f"Bearer {tok}"},
        timeout=10,
    )

    # 找最近 run
    engine = get_async_engine()
    async with AsyncSession(bind=engine) as s:
        r = await s.execute(select(PipelineRun).order_by(desc(PipelineRun.started_at)).limit(1))
        run = r.scalar_one()
        if run.status == "running":
            pytest.skip("mid-flight")
        db_stages = run.stages or []
        run_id = str(run.id)

    # 等待 terminal
    deadline = time.time() + 90
    while time.time() < deadline:
        r = await httpx.AsyncClient().get(
            f"{API}/pipeline/runs/{run_id}",
            headers={"Authorization": f"Bearer {tok}"}, timeout=10
        )
        if r.status_code == 200:
            api_stages = r.json().get("stages", [])
            if api_stages and all(s["status"] in ("completed", "failed", "skipped") for s in api_stages):
                break
        await asyncio.sleep(2)

    # 对比
    for api_s, db_s in zip(api_stages, db_stages):
        assert api_s["name"] == db_s["name"]
        assert api_s["status"] == db_s["status"]
        # Phase 17-03: graph_sync 可能 partical success, 不强求 records
        # 容忍 failed 状态下的 records_processed=0

import { test, expect } from '@playwright/test'

test.describe('Phase 17 端到端审计', () => {
  test('DAG 只显示 5 个核心 stage (无 timeseries)', async ({ page }) => {
    // Mock successful run
    await mockStages(page, completedRun())
    await page.goto('/pipeline')
    const count = await page.locator('.timeline-node').count()
    expect(count).toBe(5)  // 无 timeseries
  })

  test('重试按钮在 failed 状态下可用', async ({ page }) => {
    await mockStages(page, failedRun())
    await page.goto('/pipeline')
    const retryBtn = page.locator('button:has-text("重试")').first()
    if (await retryBtn.count() > 0) {
      await retryBtn.click()
      // 不应看到 "没有可重试的运行"
      const warn = await page.locator('text=没有可重试的运行').count()
      expect(warn).toBe(0)
    }
  })

  test('graph_sync 失败显示用户友好错误', async ({ page }) => {
    await mockStages(page, failedRun())
    await page.goto('/pipeline')
    const text = await page.evaluate(() => document.body.innerText)
    // 不应含 "Traceback" 或 raw 错误
    expect(text).not.toContain('Traceback')
    // 应含人类可读 hint
    expect(text).toMatch(/图谱|JD|Neo4j|建议/)
  })
})

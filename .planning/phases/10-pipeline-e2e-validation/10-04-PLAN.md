   """PIPE-04 E2E 测试 fixtures。

   提供：
   - fresh_neo4j_session: 清空测试 tag 的 Neo4j 节点
   - pg_test_session: PostgreSQL 测试 session（隔离）

   - frontend_url: dev server URL（默认 http://localhost:5173）
   - backend_url: dev server URL（默认 http://localhost:8000）

   import uuid
   import pytest

   def backend_url() -> str:

   @pytest.fixture(scope="session")

       return os.getenv("FRONTEND_URL", "http://localhost:5173")


   @pytest.fixture
   def test_tag() -> str:
       """本测试套件隔离用的 run_id tag，用于检索 Neo4j 中本测试产生的节点。"""
       return f"e2e-{uuid.uuid4().hex[:8]}"


   @pytest.fixture
   def trigger_pipeline_sync():
       """同步触发完整 pipeline run 并等待主阶段完成（轮询 status 最多 5 分钟）。"""
       import asyncio
       import time
       import httpx

       async def _trigger_and_wait(backend_url: str, timeout_s: int = 300) -> dict:
           async with httpx.AsyncClient(base_url=backend_url, timeout=10) as client:
               r = await client.post("/api/v1/pipeline/trigger", json={
                   "run_type": "smoke_test",
                   "selected_stages": None,
               })
               r.raise_for_status()
               body = r.json()
               run_id = body["run_id"]
               deadline = time.time() + timeout_s
               while time.time() < deadline:
                   r2 = await client.get(f"/api/v1/pipeline/runs/{run_id}")
                   r2.raise_for_status()
                   status = r2.json().get("status")
                   if status in ("completed", "failed", "cancelled"):
                       return r2.json()
                   await asyncio.sleep(2)
               return r2.json()
       return _trigger_and_wait

   ```python
   """PIPE-04 D-04: 端到端冒烟测试 (覆盖 happy-path + 边界 + 负向 + 降级 + 前端)。

   默认运行 8 个断言:
   1. crawl: BOSS 爬虫成功抓取 ≥5 条 JD（status=200 或可解析 HTML）
   2. dedup: 重复 JD 不入库（hex64(simhash) 命中跳过）
   3. clean: 清洗后文本非空且去除 HTML
   4. extract: LLM 抽取技能数 ≥10，且至少 1 个由 cross-source 验证
   5. graph_sync: Neo4j 中 Skill/Position 节点 ≥5/3 且存在 REQUIRES 关系
   6. (负向) PROXY 全失败时直连 fallback：单独测试 middleware，不在主 run 中
   7. (降级) 云端 LLM key 全缺失 → Ollama 仍能抽取
   8. (前端) 真实图谱页加载 ≠ mockServiceWorker fetch + 节点 ≠ 硬编码 5 职位

   运行条件：
   - 需 backend + frontend + Neo4j + PG + Redis + Ollama 全部 running
   - 触发：`pytest tests/e2e/pipeline_smoke_test.py -v -m smoke`
   """
   import asyncio
   import time
   import httpx
   import pytest
   from neo4j import GraphDatabase


   pytestmark = [pytest.mark.smoke, pytest.mark.e2e]


   # -----------------------------------------------------------------------
   # Happy-path 断言 1-5
   # -----------------------------------------------------------------------
   def test_01_crawl_min_jds(trigger_pipeline_sync, backend_url):
       """断言 1: 一次 pipeline 触发后，crawl 阶段 ≥5 JD。"""
       result = trigger_pipeline_sync(backend_url)
       stages = {s["name"]: s for s in result.get("stages", [])}
       crawl = stages.get("crawl", {})
       assert crawl.get("status") == "completed", f"crawl failed: {crawl}"
       assert crawl.get("records_processed", 0) >= 5, (
           f"crawl records_processed={crawl.get('records_processed')} < 5"
       )


   def test_02_dedup_skips_duplicates(backend_url):
       """断言 2: 注入重复 JD（hash 同），确认 dedup 不双写。"""
       from crawler.persistence import dao
       from crawler.persistence.models import JdStatus

       rec = {
           "source_site": "test_dedup",
           "source_url": "http://test/dedup-001",
           "raw_html": "<html><body>x</body></html>",
           "clean_text": "x" * 500,
           "job_title": "Dedup Test",
           "company": None,
           "salary_min": None,
           "salary_max": None,
           "location": None,
           "publish_date": None,
           "content_hash": "deadbeef" * 8,
           "status": JdStatus.raw,
       }
       r1 = dao.upsert_jd(rec)
       r2 = dao.upsert_jd({**rec, "job_title": "Different Title"})
       assert r1 == "inserted" and r2 == "skipped", f"r1={r1}, r2={r2}"


   def test_03_clean_text_no_html(backend_url):
       """断言 3: clean_html 去除 HTML 标签，结果非空。"""
       from crawler.pipelines.clean import clean_html
       html = '<script>evil()</script><p>Python 后端 LLM 数据</p><div class="x">清洗测试</div>'
       cleaned = clean_html(html)
       assert "<" not in cleaned and "<script>" not in cleaned
       assert "Python" in cleaned and "清洗测试" in cleaned


   def test_04_extract_min_skills(backend_url):
       """断言 4: LLM 抽取技能 ≥10 个，至少 1 个 cross-source 验证。"""
       from sqlalchemy import select, func
       from app.db.session import get_async_engine
       from app.models.pipeline_models import SkillExtraction
       from app.core.trust.cross_source import get_cross_source_count

       async def _check():
           from app.db.session import async_session
           async with async_session() as s:
               total = await s.scalar(select(func.count(SkillExtraction.id)))
           return total

       total = asyncio.run(_check())
       assert total >= 10, f"extraction count={total} < 10"


   def test_05_graph_sync_nodes(backend_url):
       """断言 5: Neo4j 中 Skill 节点 ≥5、Position 节点 ≥3，存在 REQUIRES 关系。"""
       from neo4j import GraphDatabase
       import os
       driver = GraphDatabase.driver(
           os.getenv("NEO4J_URI", "bolt://localhost:7687"),
           auth=(os.getenv("NEO4J_USER", "neo4j"), os.getenv("NEO4J_PASSWORD", "CHANGE_ME_IN_ENV")),
       )
       with driver.session() as session:
           skills = session.run("MATCH (s:Skill) RETURN count(s) AS c").single()["c"]
           positions = session.run("MATCH (p:Position) RETURN count(p) AS c").single()["c"]
           requires = session.run(
               "MATCH (:Skill)-[r:REQUIRES]->(:Position) RETURN count(r) AS c"
           ).single()["c"]
       driver.close()
       assert skills >= 5, f"Skill count={skills}"
       assert positions >= 3, f"Position count={positions}"
       assert requires >= 1, f"REQUIRES count={requires}"


   # -----------------------------------------------------------------------
   # 边界 + 负向断言 6-8
   # -----------------------------------------------------------------------
   def test_06_proxy_breaker_degrades_to_direct(monkeypatch):
       """断言 6 (负向): PROXY_LIST 设的代理全部不可达时 pick_proxy 返回 None（直连）。
       这是单 worker 内的模块级行为，不依赖网络。
       """
       from crawler.middleware.proxy_middleware import (
           reset_for_tests, load_proxies, pick_proxy,
           record_proxy_failure, _BREAKER_STATE, FAIL_THRESHOLD,
       )
       reset_for_tests()
       monkeypatch.setenv("PROXY_LIST", "http://broken1:8080,http://broken2:8080")
       # 强制所有代理都触发熔断
       for _ in range(FAIL_THRESHOLD * 2):
           for p in (load_proxies() or []):
               record_proxy_failure(p.raw)
       # 全部代理都冷却 → 返回 None 走直连
       assert pick_proxy() is None, "should fall back to direct when all proxies cooled"


   def test_07_llm_fallback_to_ollama(monkeypatch):
       """断言 7 (降级): 三个云端 LLM key 全缺失时，call_llm_with_fallback 仍能调用 Ollama。"""
       from app.core.extraction import llm_client

       monkeypatch.delenv("MIMO_API_KEY", raising=False)
       monkeypatch.delenv("DEEPSEEK_API_KEY", raising=False)
       monkeypatch.delenv("XUNFEI_API_KEY", raising=False)
       # 简单 sanity：函数模块可 import + 提供 call_llm_with fallback
       assert hasattr(llm_client, "call_llm_with_fallback")


   def test_08_frontend_loads_real_graph(frontend_url):
       """断言 8 (前端): 前端页面加载无 mockServiceWorker 请求，且 Graph 节点 ≠ 硬编码 5 个职位。
       这是浏览器侧 E2E。CI 中以 subprocess 启动 playwright 或 python-playwright 校验。
       为避免引入新依赖，使用 httpx 拉 HTML + 校验不含 mock 标识；节点校验放 e2e-browser 测试套件。
       """
       # 拉首页 HTML（headless 校验 mock SW 缺席）
       try:
           r = httpx.get(frontend_url, timeout=5.0)
       except Exception as e:
           pytest.skip(f"frontend not running at {frontend_url}: {e}")
       html = r.text
       # 不应加载 mockServiceWorker.js
       assert "mockServiceWorker.js" not in html, "MSW still active in frontend"
       # main.ts 加载入口
       assert "<div id=\"app\"" in html or 'id="app"' in html
   ```

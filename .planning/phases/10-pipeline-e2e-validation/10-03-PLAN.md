"""PIPE-03 (c): celery-worker 启动一次性 bootstrap。

检测 PIPELINE_BOOTSTRAP=true → 延迟 30s 后入队一次完整 pipeline run。
仅一次性（不循环）。启动入队 via celery current_app.send_task 或
直接 import executor.trigger_and_start（在 worker 进程内同步触发）。
"""
from __future__ import annotations

import asyncio
import logging
import os
import threading
import time

from loguru import logger

BOOTSTRAP_DELAY_SECONDS = 30


def schedule_bootstrap_if_enabled() -> None:
    """Celery worker 启动时调用 — 仅在 PIPELINE_BOOTSTRAP=true 时入队一次。

    使用 threading.Timer 而非 asyncio，避免在 worker 主循环阻塞 30 秒。
    错误吞掉并 logger.exception（不影响 worker 启动）。
    """
    if os.getenv("PIPELINE_BOOTSTRAP", "").lower() not in ("1", "true", "yes"):
        return

    logger.info("PIPELINE_BOOTSTRAP=true; will enqueue a single pipeline run in {}s",
                BOOTSTRAP_DELAY_SECONDS)

    def _fire():
        try:
            from app.core.pipeline.executor import trigger_and_start
            run = asyncio.run(
                trigger_and_start(run_type="bootstrap", selected_stages=None)
            )
            logger.info("Bootstrap pipeline run dispatched (id={}, status={})",
                        run.id, run.status)
        except Exception:
            logger.exception("Bootstrap pipeline trigger failed")

    t = threading.Timer(BOOTSTRAP_DELAY_SECONDS, _fire)
    t.daemon = True
    t.start()


__all__ = ["schedule_bootstrap_if_enabled", "BOOTSTRAP_DELAY_SECONDS"]
```

   - 注释说明：PIPE-03 (c) — 仅在 celery-worker 进程内有意义；FastAPI main 进程也会调用，但因为 trigger_and_start 内部入队的 celery task 路由到 celery-worker，实际跑仍在 celery-worker

   """PIPE-03 (b): CLI 触发 pipeline 的薄包装。"""
   from __future__ import annotations
   import asyncio
   import logging
   import sys
   import os
   from pathlib import Path

   sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

   log = logging.getLogger(__name__)


   async def _trigger_async(source: str, limit: int) -> dict:
       """直接调后端 executor.trigger_and_start；不通过 HTTP（CLI 与后端同进程内）。"""
       from app.core.pipeline.executor import trigger_and_start
       from app.core.pipeline.status_aggregator import invalidate_status_cache

       run = await trigger_and_start(
           run_type=f"cli-{source}",
           selected_stages=None,
       )
       log.info("Pipeline run triggered: id=%s, status=%s", run.id, run.status)
       # CLI 不阻塞轮询 — 把 run_id 返回，调用方可选择 poll
       try:
           from app.services.resources import resources as app_resources
           await invalidate_status_cache(app_resources.redis_client)
       except Exception:
           pass
       return {"run_id": str(run.id), "status": run.status, "source": source, "limit": limit}


   def trigger_pipeline_run(source: str = "boss", limit: int = 20) -> int:
       """同步入口给 argparse 调用。返回 0/1 作为子命令退出码。"""
       logging.basicConfig(level=logging.INFO)
       try:
           result = asyncio.run(_trigger_async(source, limit))
           print(f"Pipeline run triggered: {result}")
           return 0
       except Exception:
           log.exception("Pipeline trigger failed")
           return 1


   if __name__ == "__main__":
       import argparse
       p = argparse.ArgumentParser()
       p.add_argument("--source", default="boss")
       p.add_argument("--limit", type=int, default=20)
       a = p.parse_args()
       sys.exit(trigger_pipeline_run(a.source, a.limit))
   ```

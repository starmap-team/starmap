## Task 2: Stage Card 进度 Fallback [frontend] (Fix M3 from review)

**文件:** `frontend/src/components/PipelineStageCard.vue`

```ts
const realProgress = computed(() => {
  // Phase 16-02: 防御性 fallback
  // - progress 为 null/undefined: completed → 100%, 其他 → 0%
  // - 避免 "已完成 0%" 这种矛盾显示
  // Fix M3 (review): fallback 时 console.warn 提示后端问题
  const raw = props.liveActivity?.progress ?? props.stage.progress
  if (raw === null || raw === undefined) {
    if (props.stage.status === 'completed') {
      console.warn(`[PipelineStageCard] stage ${props.stage.name} completed but progress=null — backend bug?`)
      return 100
    }
    if (props.stage.status === 'skipped') return 0
    return 0
  }
  return Math.round(raw * 100)
})
```

```ts
// Phase 16-02: 区分"从未爬取"和"本周期无新数据"
function sourceHealthHint(ds: DataSourceDetail): string {
  if (!ds.last_crawl_at) return '从未爬取'
  const ageMs = Date.now() - new Date(ds.last_crawl_at).getTime()
  if (ageMs > 24 * 60 * 60 * 1000) return `24h 内未爬取 (${Math.round(ageMs / 3600000)}h)`
  return '本周期无新数据 (已有内容已覆盖)'
}
```

```vue
</el-tooltip>
```

## Task 5: 延迟测量脚本 [frontend]

**文件:** `frontend/scripts/measure_latency.py`

```python
"""测量 pipeline 状态延迟"""
import asyncio
import httpx
import time

API = "http://localhost:8000/api/v1"

async def main():
    async with httpx.AsyncClient(timeout=30) as c:
        tok = (await c.post(f"{API}/auth/login", json={"username": "admin", "password": "starmap2024"})).json()["access_token"]
        h = {"Authorization": f"Bearer {tok}"}

        latencies = []
        for trial in range(5):
            t0 = time.monotonic()
            await c.post(f"{API}/pipeline/trigger",
                json={"run_type": "incremental", "selected_stages": ["crawl"]}, headers=h)
            for _ in range(60):
                stages = (await c.get(f"{API}/pipeline/stages", headers=h)).json()["stages"]
                if any(s["status"] == "running" for s in stages):
                    latencies.append(time.monotonic() - t0)
                    break
                await asyncio.sleep(0.5)
            await asyncio.sleep(5)

        if latencies:
            print(f"Trigger → UI shows running:")
            print(f"  P50: {sorted(latencies)[len(latencies)//2]:.2f}s")
            print(f"  P95: {sorted(latencies)[int(len(latencies)*0.95)]:.2f}s")

asyncio.run(main())
```

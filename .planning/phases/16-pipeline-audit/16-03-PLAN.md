## Task 1: 跨端一致性抽样测试 [backend]

**文件:** `backend/tests/integration/test_cross_tier_consistency.py`

```python
"""跨端一致性 — API / DB / SSE 事件 抽样 20 次"""
import asyncio
import json
import pytest
from sqlalchemy import select, desc
from app.db.session import get_async_engine
from app.models.pipeline_models import PipelineRun
from sqlalchemy.ext.asyncio import AsyncSession
import httpx

API = "http://localhost:8000/api/v1"

@pytest.mark.parametrize("trial", range(20))
@pytest.mark.asyncio
async def test_api_matches_db_per_stage(trial):
    """抽样 N=20: API /pipeline/stages 返回的每个 stage 字段 必须与 DB PipelineRun.stages JSON 完全一致。

    Fix L4 (review): 仅在 run 已进入 terminal 状态 (completed/failed/cancelled) 时做对比。
    避免 mid-run 状态不一致导致误报。
    """
    async with httpx.AsyncClient(timeout=30) as c:
        tok = (await c.post(f"{API}/auth/login", json={"username": "admin", "password": "starmap2024"})).json()["access_token"]
        h = {"Authorization": f"Bearer {tok}"}

        # 1. API 数据
        api_stages = (await c.get(f"{API}/pipeline/stages", headers=h)).json()["stages"]
        assert api_stages, "API 返回空 stages"

        # 2. DB 数据
        engine = get_async_engine()
        async with AsyncSession(bind=engine) as s:
            db_run = (await s.execute(
                select(PipelineRun).order_by(desc(PipelineRun.started_at)).limit(1)
            )).scalar_one()
            db_stages = db_run.stages or []
            run_id = str(db_run.id)

        # Fix L4: skip if run is mid-flight
        if db_run.status == "running":
            pytest.skip(f"Run {run_id} still running — cross-tier test needs terminal state")

        # 3. 字段对比
        for api_s, db_s in zip(api_stages, db_stages):
            assert api_s["name"] == db_s["name"], f"Name mismatch: {api_s['name']} vs {db_s['name']}"
            assert api_s["status"] == db_s["status"], \
                f"Status mismatch for {api_s['name']}: API={api_s['status']} DB={db_s['status']}"
            assert api_s["records_processed"] == db_s.get("records_processed", 0)
            assert abs(api_s.get("progress") or 0 - (db_s.get("progress") or 0)) < 0.01
            assert api_s.get("errors", []) == db_s.get("errors", []), \
                f"Errors mismatch for {api_s['name']}"
            assert api_s.get("duration_ms") == db_s.get("duration_ms")
```

## Task 2: 性能瓶颈分析 [backend]

**测量目标:** `/pipeline/stages` 响应时间

```python
"""performance test"""
import asyncio
import httpx
import time

API = "http://localhost:8000/api/v1"

async def measure():
    async with httpx.AsyncClient(timeout=10) as c:
        tok = (await c.post(f"{API}/auth/login", json={"username": "admin", "password": "starmap2024"})).json()["access_token"]
        h = {"Authorization": f"Bearer {tok}"}

        latencies = []
        for _ in range(100):
            t0 = time.monotonic()
            await c.get(f"{API}/pipeline/stages", headers=h)
            latencies.append(time.monotonic() - t0)

        latencies.sort()
        print(f"P50:  {latencies[50]*1000:.1f}ms")
        print(f"P95:  {latencies[95]*1000:.1f}ms")
        print(f"P99:  {latencies[99]*1000:.1f}ms")
```

**预期发现:**
- /pipeline/stages 执行 1 次 SQL (select PipelineRun)
- P95 应该 < 50ms（纯 SQL 不复杂）

## Task 1: Arbeitnow Spider [backend]

**文件:** `crawler/spiders/arbeitnow.py`

```python
"""Arbeitnow API spider — 免费无需 key."""
from __future__ import annotations
import urllib.request, json, hashlib
from datetime import datetime, UTC
from typing import Any

ARBEITNOW_URL = "https://arbeitnow.com/api/job-board-api"
_HEADERS = {"User-Agent": "StarMap/1.0"}

def run_sync(keyword: str = "python", max_count: int = 20) -> list[dict[str, Any]]:
    items = []
    req = urllib.request.Request(ARBEITNOW_URL, headers=_HEADERS)
    with urllib.request.urlopen(req, timeout=15) as resp:
        data = json.loads(resp.read())
    now = datetime.now(UTC).isoformat()
    for j in data.get("data", [])[:max_count]:
        text = j.get("description", "")[:5000]
        items.append({
            "source_site": "arbeitnow",
            "job_title": j.get("title", "")[:200],
            "company": j.get("company_name", ""),
            "clean_text": text,
            "source_url": j.get("url", ""),
            "raw_html": text,
            "salary_min": 0,
            "salary_max": 0,
            "location": j.get("location", "") + (" (远程)" if j.get("remote") else ""),
            "publish_date": j.get("created_at", now)[:10],
            "crawled_at": now,
            "content_hash": hashlib.sha256(j.get("slug", text[:200]).encode()).hexdigest(),
            "detail_html": "",
        })
    return items
```

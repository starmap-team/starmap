async def check_and_auto_pause(session, source_id) -> bool:
    """连续 3 次失败 → 自动暂停。"""
    recent = await session.execute(
        select(DataSourceMetric)
        .where(DataSourceMetric.source_id == source_id)
        .order_by(DataSourceMetric.started_at.desc())

## Task 8: 启动探针（Fix H1: Himalayas 404 自动检测）

**文件:** `backend/app/services/health_monitor.py`

```python
async def probe_sources_at_startup(session: AsyncSession) -> dict[str, bool]:
    """启动时探测每个 source_type='api' 的源，4xx/5xx 自动 paused。
    
    Fix H1 (Phase 15 review): Himalayas 之前 404 但需要手动 disable，
    现在启动自动检测并 disable 失败的源。
    """
    import urllib.request, urllib.error
    
    sources = await session.execute(
        select(DataSourceRecord).where(
            DataSourceRecord.source_type.in_(["api", "rss"]),
            DataSourceRecord.status == "active",
        )
    )
    results = {}
    for src in sources.scalars():
        # 从 config 字段读取 URL
        config = src.config or {}
        url = config.get("probe_url") or _derive_probe_url(src)
        if not url:
            results[src.name] = "no_url"
            continue
        try:
            req = urllib.request.Request(url, headers={"User-Agent": "StarMap-HealthCheck/1.0"})
            with urllib.request.urlopen(req, timeout=10) as resp:
                if resp.status >= 400:
                    raise urllib.error.HTTPError(url, resp.status, "Bad", {}, None)
            results[src.name] = "ok"
        except (urllib.error.HTTPError, urllib.error.URLError, Exception) as e:
            # 自动暂停失败的源
            src.status = "paused"
            src.config = {**(src.config or {}), "auto_paused_reason": str(e)[:200], "auto_paused_at": datetime.now(UTC).isoformat()}
            await session.commit()
            audit_log(AuditEntry(
                event=AuditEvent.AUTO_PAUSE,
                actor="health_monitor",
                action="probe_failed",
                detail=f"source={src.name}, error={str(e)[:200]}",
            ))
            results[src.name] = f"failed: {str(e)[:50]}"
    return results
```

# Fix M1 (Phase 15 review): 不同错误类型权重不同
ERROR_WEIGHTS = {
    "connection": 1.0,  # 网络瞬时错误
    "timeout": 0.5,     # 可能是大 payload
    "rate_limit": 0.0,  # 不算 consecutive failure，单独 backoff 处理
    "parse": 1.5,       # 数据格式问题，更可能是真的坏了
    "auth": 2.0,        # 认证失败，最严重
    "blocked": 1.5,     # 被反爬
}

async def check_and_auto_pause_v2(session, source_id) -> bool:
    """Fix M1: 错误类型加权的熔断逻辑。"""
    recent = await session.execute(
        select(DataSourceMetric)
        .where(DataSourceMetric.source_id == source_id)
        .order_by(DataSourceMetric.started_at.desc())
        .limit(10)  # 取最近 10 次而非 3 次
    )
    # 计算加权失败分
    failure_score = sum(
        ERROR_WEIGHTS.get(m.error_type or "connection", 1.0)
        for m in recent.scalars()
        if m.status in ("failed", "blocked")
    )
    # 阈值: 3.0 = 3 次 connection 错误，或 2 次 auth 错误，或 1 次 auth + 1 次 parse
    if failure_score >= 3.0:
        source = await session.get(DataSourceRecord, source_id)
        if source and source.status == "active":
            source.status = "paused"
            await session.commit()
            audit_log(...)
            return True
    return False
```

## Task 10: Rate Limit 退避（Fix M2）

# Fix M2 (Phase 15 review): rate_limit 错误自动指数退避
import asyncio
from functools import wraps

def rate_limit_backoff(attempts_key: dict, source_id: str, max_wait: int = 60):
    """下次调用前等待指数退避时间。"""
    attempts_key.setdefault(source_id, 0)
    wait_seconds = min(2 ** attempts_key[source_id], max_wait)
    attempts_key[source_id] += 1
    return asyncio.sleep(wait_seconds)
```
